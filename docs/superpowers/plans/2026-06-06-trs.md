```python
"""SignalRepository Protocol + in-memory implementation (hermetic tests).

Postgres impl lives in repository_pg.py; both satisfy this Protocol. `_active` mirrors the SQL
temporal predicate exactly (the MAR parity lesson): an open-ended alias is NOT active before its
valid_from when a past valid_at is supplied.
"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from typing import Any, Protocol, runtime_checkable
from uuid import UUID

from rca_contracts import SignalDescriptor


@dataclass(frozen=True)
class AliasRow:
    signal_id: UUID
    tenant_id: UUID
    source_system: str
    raw_tag: str
    raw_unit: str
    valid_from: datetime
    valid_to: datetime | None
    mapping_source: str
    confidence: float


@runtime_checkable
class SignalRepository(Protocol):
    async def find_active_aliases(self, tenant: UUID, source: str, raw_tag: str,
                                  *, valid_at: datetime | None) -> list[AliasRow]: ...
    async def get_signal(self, tenant: UUID, signal_id: UUID) -> SignalDescriptor | None: ...
    async def search_signals(self, tenant: UUID, *, asset_id: UUID | None = None,
                             role: str | None = None, role_pattern: str | None = None,
                             limit: int = 50) -> list[SignalDescriptor]: ...
    async def get_alias_for(self, tenant: UUID, signal_id: UUID, source: str) -> AliasRow | None: ...
    async def upsert_unresolved(self, tenant: UUID, source: str, raw_tag: str) -> None: ...
    async def upsert_signal(self, signal: SignalDescriptor) -> None: ...
    async def upsert_alias(self, alias: AliasRow) -> None: ...


def _active(alias: AliasRow, valid_at: datetime | None) -> bool:
    if valid_at is None:
        return alias.valid_to is None
    if alias.valid_from > valid_at:
        return False
    return alias.valid_to is None or valid_at < alias.valid_to


class InMemoryRepository:
    def __init__(self) -> None:
        self.signals: dict[tuple[UUID, UUID], SignalDescriptor] = {}
        self.aliases: list[AliasRow] = []
        self.unresolved: dict[tuple[UUID, str, str], dict[str, Any]] = {}

    async def upsert_signal(self, signal: SignalDescriptor) -> None:
        self.signals[(signal.tenant_id, signal.signal_id)] = signal

    async def upsert_alias(self, alias: AliasRow) -> None:
        self.aliases = [a for a in self.aliases
                        if not (a.tenant_id == alias.tenant_id
                                and a.source_system == alias.source_system
                                and a.raw_tag == alias.raw_tag and a.valid_to is None)]
        self.aliases.append(alias)

    async def find_active_aliases(self, tenant, source, raw_tag, *, valid_at):
        return [a for a in self.aliases
                if a.tenant_id == tenant and a.source_system == source
                and a.raw_tag == raw_tag and _active(a, valid_at)]

    async def get_signal(self, tenant, signal_id):
        return self.signals.get((tenant, signal_id))

    async def search_signals(self, tenant, *, asset_id=None, role=None, role_pattern=None, limit=50):
        out = []
        for s in self.signals.values():
            if s.tenant_id != tenant:
                continue
            if asset_id and s.asset_id != asset_id:
                continue
            if role and s.role != role:
                continue
            if role_pattern and role_pattern.replace("%", "") not in s.role:
                continue
            out.append(s)
        return out[:limit]

    async def get_alias_for(self, tenant, signal_id, source):
        for a in self.aliases:
            if (a.tenant_id == tenant and a.signal_id == signal_id
                    and a.source_system == source and a.valid_to is None):
                return a
        return None

    async def upsert_unresolved(self, tenant, source, raw_tag):
        key = (tenant, source, raw_tag)
        row = self.unresolved.get(key)
        if row:
            row["occurrence_count"] += 1
        else:
            self.unresolved[key] = {"occurrence_count": 1}


__all__ = ["SignalRepository", "AliasRow", "InMemoryRepository"]
```

```python
"""The 4-step tag-resolution algorithm (SPEC-003). Pure logic over a SignalRepository."""
from __future__ import annotations

import re
from dataclasses import dataclass, field
from datetime import datetime
from uuid import UUID

from rca_contracts import ResolveStatus

from .repository import AliasRow, SignalRepository

_REGEX_CONFIDENCE = 0.70


@dataclass(frozen=True)
class TagResolution:
    status: ResolveStatus
    signal_id: UUID | None
    confidence: float
    mapping_source: str
    alternatives: list[UUID] = field(default_factory=list)


def _extract(raw_tag: str, patterns: list[str]) -> dict[str, str]:
    for pat in patterns:
        m = re.search(pat, raw_tag)
        if m:
            return {k: v for k, v in m.groupdict().items() if v}
    return {}


async def resolve_tag(
    repo: SignalRepository,
    raw_tag: str,
    source: str,
    tenant: UUID,
    *,
    asset_hint: UUID | None = None,
    valid_at: datetime | None = None,
    min_confidence: float = 0.85,
    regex_patterns: list[str] | None = None,
) -> TagResolution:
    async def _signals_of(aliases: list[AliasRow]) -> dict[UUID, AliasRow]:
        return {a.signal_id: a for a in aliases}

    # Step 1: exact active-alias match
    exact = await repo.find_active_aliases(tenant, source, raw_tag, valid_at=valid_at)
    by_signal = await _signals_of(exact)
    if len(by_signal) == 1:
        a = next(iter(by_signal.values()))
        status: ResolveStatus = "resolved" if a.confidence >= min_confidence else "unresolved"
        if status == "unresolved":
            await repo.upsert_unresolved(tenant, source, raw_tag)
        return TagResolution(status, a.signal_id, a.confidence, a.mapping_source)
    if len(by_signal) > 1:
        # Step 2: asset-hinted disambiguation
        if asset_hint is not None:
            hinted = []
            for sid in by_signal:
                sig = await repo.get_signal(tenant, sid)
                if sig is not None and sig.asset_id == asset_hint:
                    hinted.append(sid)
            if len(hinted) == 1:
                a = by_signal[hinted[0]]
                return TagResolution("resolved", hinted[0], a.confidence, "asset_hinted")
        return TagResolution("ambiguous", None, max(a.confidence for a in exact), "exact",
                             alternatives=sorted(by_signal, key=str))

    # Step 3: regex heuristic -> search by (asset_hint, role)
    parsed = _extract(raw_tag, regex_patterns or [])
    role = parsed.get("role")
    if role:
        matches = await repo.search_signals(tenant, asset_id=asset_hint, role=role)
        if len(matches) == 1:
            status = "resolved" if _REGEX_CONFIDENCE >= min_confidence else "unresolved"
            if status == "unresolved":
                await repo.upsert_unresolved(tenant, source, raw_tag)
            return TagResolution(status, matches[0].signal_id, _REGEX_CONFIDENCE, "regex_heuristic")

    # Step 4: unresolved
    await repo.upsert_unresolved(tenant, source, raw_tag)
    return TagResolution("unresolved", None, 0.0, "none")


__all__ = ["TagResolution", "resolve_tag"]
```

```python
"""FastMCP server for TRS read/resolve tools (hand-wired; read TRS's own repository)."""
from __future__ import annotations

from datetime import datetime, timezone
from uuid import UUID, uuid4

from fastmcp import FastMCP
from pydantic import AwareDatetime, BaseModel
from rca_connector_sdk import NotFound, ProvenanceAccumulator, build_server, map_source_error
from rca_contracts import AssetID, ResolveTagOutput, SignalDescriptor, ToolResponse

from .repository import SignalRepository
from .resolution import resolve_tag

_VERSION = "0.1.0"
_SOURCE = "trs"


class ResolveTagRequest(BaseModel):
    raw_tag: str
    source_system: str
    asset_hint: AssetID | None = None
    time: AwareDatetime | None = None
    min_confidence: float = 0.85


class GetSignalRequest(BaseModel):
    signal_id: UUID


class SearchSignalsRequest(BaseModel):
    asset_id: AssetID | None = None
    role: str | None = None
    role_pattern: str | None = None
    limit: int = 50


def _ok(envelope, data, *, tool, source_query, record_count, raw_tags):
    prov = ProvenanceAccumulator()
    prov.record(source_query=source_query, record_count=record_count, raw_tags=raw_tags)
    provenance = prov.build(tool_name=tool, tool_version=_VERSION, source=_SOURCE,
                            queried_at=datetime.now(timezone.utc), response_id=uuid4())
    return envelope.ok(data, provenance)


def make_trs_mcp(*, repo: SignalRepository, tenant_id: UUID,
                 regex_patterns: list[str] | None = None) -> FastMCP:
    mcp = build_server("trs")

    @mcp.tool(name="trs.resolve_tag")
    async def resolve(request: ResolveTagRequest) -> ToolResponse[ResolveTagOutput]:
        envelope = ToolResponse[ResolveTagOutput]
        try:
            r = await resolve_tag(repo, request.raw_tag, request.source_system, tenant_id,
                                  asset_hint=request.asset_hint, valid_at=request.time,
                                  min_confidence=request.min_confidence, regex_patterns=regex_patterns)
            signal = await repo.get_signal(tenant_id, r.signal_id) if r.signal_id else None
            alts = [s for s in [await repo.get_signal(tenant_id, x) for x in r.alternatives] if s]
            out = ResolveTagOutput(status=r.status, signal=signal, confidence=r.confidence,
                                   mapping_source=r.mapping_source, alternatives=alts)
            return _ok(envelope, out, tool="trs.resolve_tag",
                       source_query=f"resolve {request.source_system}:{request.raw_tag}",
                       record_count=1 if signal else 0,
                       raw_tags=[f"{request.source_system}:{request.raw_tag}"])
        except Exception as exc:  # noqa: BLE001
            return envelope.fail(map_source_error(exc))

    @mcp.tool(name="trs.get_signal")
    async def get_signal(request: GetSignalRequest) -> ToolResponse[SignalDescriptor]:
        envelope = ToolResponse[SignalDescriptor]
        try:
            sig = await repo.get_signal(tenant_id, request.signal_id)
            if sig is None:
                raise NotFound(f"signal {request.signal_id} not found")
            return _ok(envelope, sig, tool="trs.get_signal",
                       source_query=f"get {request.signal_id}", record_count=1,
                       raw_tags=[str(request.signal_id)])
        except Exception as exc:  # noqa: BLE001
            return envelope.fail(map_source_error(exc))

    @mcp.tool(name="trs.search_signals")
    async def search_signals(request: SearchSignalsRequest) -> ToolResponse[list[SignalDescriptor]]:
        envelope = ToolResponse[list[SignalDescriptor]]
        try:
            sigs = await repo.search_signals(tenant_id, asset_id=request.asset_id, role=request.role,
                                             role_pattern=request.role_pattern, limit=request.limit)
            return _ok(envelope, sigs, tool="trs.search_signals",
                       source_query=f"search asset={request.asset_id} role={request.role}",
                       record_count=len(sigs), raw_tags=[s.role for s in sigs])
        except Exception as exc:  # noqa: BLE001
            return envelope.fail(map_source_error(exc))

    return mcp


__all__ = ["make_trs_mcp", "ResolveTagRequest", "GetSignalRequest", "SearchSignalsRequest"]
```

```python
"""TrsResolver — full connector_sdk SignalResolver for the signal-scoped connectors (PI, OPC UA)."""
from __future__ import annotations

from uuid import UUID

from rca_connector_sdk import SourceBinding
from rca_connector_sdk.errors import UnresolvedSignal
from rca_contracts import SignalDescriptor

from .repository import SignalRepository


class TrsResolver:
    def __init__(self, *, repo: SignalRepository, tenant_id: UUID) -> None:
        self._repo = repo
        self._tenant = tenant_id

    async def resolve(self, signal_id: UUID) -> SignalDescriptor:
        sig = await self._repo.get_signal(self._tenant, signal_id)
        if sig is None:
            raise UnresolvedSignal(f"signal {signal_id} not found in TRS")
        return sig

    async def source_binding(self, entity_id: UUID, source: str) -> SourceBinding:
        alias = await self._repo.get_alias_for(self._tenant, entity_id, source)
        if alias is None:
            raise UnresolvedSignal(f"no {source!r} alias for signal {entity_id} in TRS")
        return SourceBinding(handle=alias.raw_tag, raw_unit=alias.raw_unit)


__all__ = ["TrsResolver"]
```

```python
"""PostgresRepository — SQLAlchemy 2.0 async implementation of SignalRepository."""
from __future__ import annotations

from uuid import uuid4

from sqlalchemy import and_, select, update
from sqlalchemy.dialects.postgresql import insert as pg_insert
from sqlalchemy.ext.asyncio import async_sessionmaker

from rca_contracts import PressureReference, SignalDescriptor

from .models import Signal, SignalAlias, SignalAliasUnresolved
from .repository import AliasRow


def _to_signal(s: Signal) -> SignalDescriptor:
    return SignalDescriptor(
        signal_id=s.signal_id, tenant_id=s.tenant_id, asset_id=s.asset_id, role=s.role,
        qudt_unit=s.qudt_unit,
        pressure_reference=PressureReference(s.pressure_reference) if s.pressure_reference
        else PressureReference.not_applicable,
        range_min=s.range_min, range_max=s.range_max, description=s.description)


def _to_aliasrow(a: SignalAlias) -> AliasRow:
    return AliasRow(signal_id=a.signal_id, tenant_id=a.tenant_id, source_system=a.source_system,
                    raw_tag=a.raw_tag, raw_unit=a.raw_unit, valid_from=a.valid_from,
                    valid_to=a.valid_to, mapping_source=a.mapping_source, confidence=a.confidence)


class PostgresRepository:
    def __init__(self, session_factory: async_sessionmaker) -> None:
        self._sf = session_factory

    async def upsert_signal(self, signal: SignalDescriptor) -> None:
        async with self._sf() as s, s.begin():
            v = dict(signal_id=signal.signal_id, tenant_id=signal.tenant_id, asset_id=signal.asset_id,
                     role=signal.role, qudt_unit=signal.qudt_unit,
                     pressure_reference=signal.pressure_reference.value
                     if signal.pressure_reference else None,
                     range_min=signal.range_min, range_max=signal.range_max,
                     description=signal.description)
            stmt = pg_insert(Signal).values(**v).on_conflict_do_update(
                index_elements=[Signal.signal_id], set_={k: v[k] for k in v if k != "signal_id"})
            await s.execute(stmt)

    async def upsert_alias(self, alias: AliasRow) -> None:
        async with self._sf() as s, s.begin():
            # TRS supersede is per (tenant, source, raw_tag, SIGNAL) — the same raw_tag may map to
            # multiple signals (the ambiguous case), so we must NOT close sibling signals' aliases.
            await s.execute(update(SignalAlias).where(and_(
                SignalAlias.tenant_id == alias.tenant_id,
                SignalAlias.source_system == alias.source_system,
                SignalAlias.raw_tag == alias.raw_tag,
                SignalAlias.signal_id == alias.signal_id,
                SignalAlias.valid_to.is_(None))).values(valid_to=alias.valid_from))
            await s.execute(pg_insert(SignalAlias).values(
                alias_id=uuid4(), signal_id=alias.signal_id, tenant_id=alias.tenant_id,
                source_system=alias.source_system, raw_tag=alias.raw_tag, raw_unit=alias.raw_unit,
                valid_from=alias.valid_from, valid_to=alias.valid_to,
                mapping_source=alias.mapping_source, confidence=alias.confidence))

    async def find_active_aliases(self, tenant, source, raw_tag, *, valid_at):
        async with self._sf() as s:
            q = select(SignalAlias).where(and_(
                SignalAlias.tenant_id == tenant, SignalAlias.source_system == source,
                SignalAlias.raw_tag == raw_tag))
            if valid_at is None:
                q = q.where(SignalAlias.valid_to.is_(None))
            else:
                q = q.where(and_(SignalAlias.valid_from <= valid_at,
                                 (SignalAlias.valid_to.is_(None)) | (SignalAlias.valid_to > valid_at)))
            return [_to_aliasrow(r) for r in (await s.execute(q)).scalars()]

    async def get_signal(self, tenant, signal_id):
        async with self._sf() as s:
            q = select(Signal).where(and_(Signal.tenant_id == tenant, Signal.signal_id == signal_id))
            row = (await s.execute(q)).scalar_one_or_none()
            return _to_signal(row) if row else None

    async def search_signals(self, tenant, *, asset_id=None, role=None, role_pattern=None, limit=50):
        async with self._sf() as s:
            q = select(Signal).where(Signal.tenant_id == tenant)
            if asset_id:
                q = q.where(Signal.asset_id == asset_id)
            if role:
                q = q.where(Signal.role == role)
            if role_pattern:
                q = q.where(Signal.role.like(role_pattern))
            return [_to_signal(r) for r in (await s.execute(q.limit(limit))).scalars()]

    async def get_alias_for(self, tenant, signal_id, source):
        async with self._sf() as s:
            q = select(SignalAlias).where(and_(
                SignalAlias.tenant_id == tenant, SignalAlias.signal_id == signal_id,
                SignalAlias.source_system == source, SignalAlias.valid_to.is_(None))).limit(1)
            row = (await s.execute(q)).scalar_one_or_none()
            return _to_aliasrow(row) if row else None

    async def upsert_unresolved(self, tenant, source, raw_tag):
        async with self._sf() as s, s.begin():
            stmt = pg_insert(SignalAliasUnresolved).values(
                tenant_id=tenant, source_system=source, raw_tag=raw_tag, occurrence_count=1)
            stmt = stmt.on_conflict_do_update(
                index_elements=[SignalAliasUnresolved.tenant_id, SignalAliasUnresolved.source_system,
                                SignalAliasUnresolved.raw_tag],
                set_={"occurrence_count": SignalAliasUnresolved.occurrence_count + 1})
            await s.execute(stmt)


__all__ = ["PostgresRepository"]
```
